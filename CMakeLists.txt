cmake_minimum_required(VERSION 3.20)
project(LagSwitchWindows VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(LagSwitchTests tests/CoreTests.cpp)
target_include_directories(LagSwitchTests PRIVATE src)
add_test(NAME core COMMAND LagSwitchTests)
if(WIN32)
  enable_language(RC)
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  add_executable(LagSwitch WIN32 src/Main.cpp src/App.rc)
  target_include_directories(LagSwitch PRIVATE src)
  target_link_libraries(LagSwitch PRIVATE fwpuclnt iphlpapi ws2_32 rpcrt4 shell32 advapi32 user32 gdi32 wtsapi32)
  if(MSVC)
    # App.rc contains the elevation manifest for both MSVC and MinGW builds.
    target_link_options(LagSwitch PRIVATE /MANIFEST:NO)
  elseif(MINGW)
    target_link_options(LagSwitch PRIVATE -municode -static)
  endif()
  add_executable(LagSwitchIntegration tests/WindowsIntegration.cpp)
  target_include_directories(LagSwitchIntegration PRIVATE src)
  target_link_libraries(LagSwitchIntegration PRIVATE fwpuclnt ws2_32 rpcrt4)
  if(MINGW)
    target_link_options(LagSwitchIntegration PRIVATE -static)
  endif()
  add_test(NAME windows-packet-filter COMMAND LagSwitchIntegration)
  set_tests_properties(windows-packet-filter PROPERTIES TIMEOUT 45)
  install(TARGETS LagSwitch RUNTIME DESTINATION .)
  install(FILES README.md LICENSE DESTINATION .)
endif()
